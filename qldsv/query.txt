-- ============================================================
--  DIEM
--  Form NhapDiem.DataGrid: MSV, HoTen, DiemCC, DiemGK,
--                           DiemCK, DTB[readonly], GPA[readonly],
--                           XepLoai[readonly]
--
--  Công thức (theo thông báo trên form):
--    Điểm quá trình 40% (CC=10% + GK=30%) + Cuối kỳ 60%
--    DTB = DiemCC*0.1 + DiemGK*0.3 + DiemCK*0.6
--
--  Thang GPA 4.0 theo chuẩn Bộ GD-ĐT:
--    A  (>=8.5) = 4.0  |  B+ (>=7.0) = 3.5  |  B  (>=6.0) = 3.0
--    C+ (>=5.5) = 2.5  |  C  (>=5.0) = 2.0  |  D+ (>=4.5) = 1.5
--    D  (>=4.0) = 1.0  |  F  (<4.0)  = 0.0
-- ============================================================
CREATE TABLE Diem
(
    MaDiem          INT          NOT NULL IDENTITY(1,1),
    MaDK            INT          NOT NULL,
    DiemCC          DECIMAL(4,2) NULL,    -- Chuyên cần  (0–10)
    DiemGK          DECIMAL(4,2) NULL,    -- Giữa kỳ    (0–10)
    DiemCK          DECIMAL(4,2) NULL,    -- Cuối kỳ    (0–10)

    -- Điểm trung bình hệ 10
    DiemTB AS (
        CAST(ROUND(
            ISNULL(DiemCC, 0) * 0.1 +
            ISNULL(DiemGK, 0) * 0.3 +
            ISNULL(DiemCK, 0) * 0.6
        , 2) AS DECIMAL(4,2))
    ) PERSISTED,

    -- Điểm GPA hệ 4
    DiemGPA AS (
        CASE
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 8.5 THEN CAST(4.0 AS DECIMAL(3,1))
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 7.0 THEN CAST(3.5 AS DECIMAL(3,1))
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 6.0 THEN CAST(3.0 AS DECIMAL(3,1))
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 5.5 THEN CAST(2.5 AS DECIMAL(3,1))
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 5.0 THEN CAST(2.0 AS DECIMAL(3,1))
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 4.5 THEN CAST(1.5 AS DECIMAL(3,1))
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 4.0 THEN CAST(1.0 AS DECIMAL(3,1))
            ELSE CAST(0.0 AS DECIMAL(3,1))
        END
    ) PERSISTED,

    -- Xếp loại chữ
    XepLoai AS (
        CASE
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 8.5 THEN 'A'
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 7.0 THEN 'B+'
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 6.0 THEN 'B'
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 5.5 THEN 'C+'
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 5.0 THEN 'C'
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 4.5 THEN 'D+'
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 4.0 THEN 'D'
            ELSE 'F'
        END
    ) PERSISTED,

    -- Kết quả
    KetQua AS (
        CASE
            WHEN ISNULL(DiemCC,0)*0.1+ISNULL(DiemGK,0)*0.3+ISNULL(DiemCK,0)*0.6 >= 4.0
            THEN 'Dat' ELSE 'KhongDat'
        END
    ) PERSISTED,

    DaChotDiem    BIT          NOT NULL DEFAULT 0,
    NgayNhap      DATETIME     NOT NULL DEFAULT GETDATE(),
    MaGV_NhapDiem VARCHAR(10)  NULL,

    CONSTRAINT PK_Diem            PRIMARY KEY (MaDiem),
    CONSTRAINT FK_Diem_DangKy     FOREIGN KEY (MaDK)           REFERENCES DangKyHocPhan(MaDK),
    CONSTRAINT FK_Diem_GiangVien  FOREIGN KEY (MaGV_NhapDiem)  REFERENCES GiangVien(MaGV),
    CONSTRAINT UQ_Diem_MaDK       UNIQUE (MaDK),
    CONSTRAINT CK_Diem_CC         CHECK (DiemCC IS NULL OR DiemCC BETWEEN 0 AND 10),
    CONSTRAINT CK_Diem_GK         CHECK (DiemGK IS NULL OR DiemGK BETWEEN 0 AND 10),
    CONSTRAINT CK_Diem_CK         CHECK (DiemCK IS NULL OR DiemCK BETWEEN 0 AND 10)
);
GO

-- ============================================================
--  VIEWS
--  QUY TẮC: tên cột trong view PHẢI KHỚP với tên cột
--  mà stored procedure sẽ tham chiếu.
--  KHÔNG đổi alias nếu không cần thiết.
-- ============================================================
CREATE VIEW vw_BangDiem
AS
SELECT
    sv.MaSV,
    sv.HoTen,
    l.MaLop,
    l.TenLop,
    k.TenKhoa,
    hk.MaHK,
    hk.TenHK,
    hk.NamHoc,
    mh.MaMH,
    mh.TenMH,
    mh.SoTinChi,
    d.DiemCC,
    d.DiemGK,
    d.DiemCK,
    d.DiemTB,
    d.DiemGPA,
    d.XepLoai,
    d.KetQua,
    d.DaChotDiem,
    d.MaDiem,
    dk.MaDK
FROM SinhVien sv
JOIN Lop l                ON sv.MaLop  = l.MaLop
JOIN Khoa k               ON l.MaKhoa  = k.MaKhoa
JOIN DangKyHocPhan dk     ON sv.MaSV   = dk.MaSV
JOIN LopHocPhan lhp       ON dk.MaLHP  = lhp.MaLHP
JOIN MonHoc mh            ON lhp.MaMH  = mh.MaMH
JOIN HocKy hk             ON lhp.MaHK  = hk.MaHK
LEFT JOIN Diem d          ON dk.MaDK   = d.MaDK;
GO

CREATE VIEW vw_LopHocPhan
AS
SELECT
    lhp.MaLHP,
    lhp.MaMH,
    mh.TenMH,
    mh.SoTinChi,
    lhp.MaGV,
    gv.HoTen         AS HoTenGV,
    gv.HocVi,
    hk.MaHK,
    hk.TenHK,
    hk.NamHoc,
    lhp.Phong,
    lhp.LichHoc,
    lhp.SiSoHienTai,          -- giữ nguyên tên cột, không đổi alias
    lhp.SiSoToiDa,
    lhp.TrangThai
FROM LopHocPhan lhp
JOIN MonHoc  mh ON lhp.MaMH = mh.MaMH
JOIN GiangVien gv ON lhp.MaGV = gv.MaGV
JOIN HocKy  hk  ON lhp.MaHK  = hk.MaHK;
GO

CREATE VIEW vw_NhapDiem
AS
SELECT
    dk.MaDK,
    dk.MaLHP,
    sv.MaSV,
    sv.HoTen,
    d.MaDiem,
    d.DiemCC,
    d.DiemGK,
    d.DiemCK,
    d.DiemTB,
    d.DiemGPA,
    d.XepLoai,
    d.KetQua,
    d.DaChotDiem,
    d.NgayNhap
FROM DangKyHocPhan dk
JOIN SinhVien sv       ON dk.MaSV  = sv.MaSV
LEFT JOIN Diem d       ON dk.MaDK  = d.MaDK;
GO

CREATE VIEW vw_GPATichLuy
AS
SELECT
    sv.MaSV,
    sv.HoTen,
    sv.TinhTrang,
    l.MaLop,
    l.TenLop,
    k.MaKhoa,
    k.TenKhoa,
    ROUND(
        SUM(CAST(mh.SoTinChi AS FLOAT) * CAST(d.DiemGPA AS FLOAT))
        / NULLIF(SUM(mh.SoTinChi), 0)
    , 2)                 AS GPATichLuy,
    SUM(mh.SoTinChi)     AS TongTinChi,
    COUNT(dk.MaDK)       AS SoMonDaHoc
FROM SinhVien sv
JOIN Lop l              ON sv.MaLop  = l.MaLop
JOIN Khoa k             ON l.MaKhoa  = k.MaKhoa
JOIN DangKyHocPhan dk   ON sv.MaSV   = dk.MaSV
JOIN LopHocPhan lhp     ON dk.MaLHP  = lhp.MaLHP
JOIN MonHoc mh          ON lhp.MaMH  = mh.MaMH
JOIN Diem d             ON dk.MaDK   = d.MaDK
WHERE dk.TrangThai = 'HoanThanh'
  AND d.DiemGPA    IS NOT NULL
GROUP BY sv.MaSV, sv.HoTen, sv.TinhTrang,
         l.MaLop, l.TenLop, k.MaKhoa, k.TenKhoa;
GO

CREATE VIEW vw_ThongKe_PhanBoDiem
AS
SELECT
    hk.MaHK,
    hk.TenHK,
    hk.NamHoc,
    k.MaKhoa,
    k.TenKhoa,
    mh.MaMH,
    mh.TenMH,
    d.XepLoai,
    d.KetQua,
    COUNT(*) AS SoLuong
FROM Diem d
JOIN DangKyHocPhan dk ON d.MaDK     = dk.MaDK
JOIN LopHocPhan lhp   ON dk.MaLHP   = lhp.MaLHP
JOIN MonHoc mh        ON lhp.MaMH   = mh.MaMH
JOIN Khoa k           ON mh.MaKhoa  = k.MaKhoa
JOIN HocKy hk         ON lhp.MaHK   = hk.MaHK
WHERE dk.TrangThai = 'HoanThanh'
GROUP BY hk.MaHK, hk.TenHK, hk.NamHoc,
         k.MaKhoa, k.TenKhoa,
         mh.MaMH, mh.TenMH,
         d.XepLoai, d.KetQua;
GO

CREATE VIEW vw_ThongKe_DatRot
AS
SELECT
    lhp.MaHK,
    mh.MaMH,
    mh.TenMH,
    SUM(CASE WHEN d.KetQua = 'Dat'      THEN 1 ELSE 0 END) AS SoDat,
    SUM(CASE WHEN d.KetQua = 'KhongDat' THEN 1 ELSE 0 END) AS SoRot,
    ROUND(AVG(CAST(d.DiemTB AS FLOAT)), 2)                  AS DiemTBMon
FROM Diem d
JOIN DangKyHocPhan dk ON d.MaDK    = dk.MaDK
JOIN LopHocPhan lhp   ON dk.MaLHP  = lhp.MaLHP
JOIN MonHoc mh        ON lhp.MaMH  = mh.MaMH
WHERE dk.TrangThai = 'HoanThanh'
GROUP BY lhp.MaHK, mh.MaMH, mh.TenMH;
GO

CREATE VIEW vw_Dashboard_Admin
AS
SELECT
    (SELECT COUNT(*) FROM SinhVien
     WHERE TinhTrang = 'DangHoc')                                  AS TongSV,
    (SELECT COUNT(*) FROM LopHocPhan
     WHERE TrangThai = 'DangMo')                                   AS LopHPDangMo,
    (SELECT COUNT(DISTINCT MaMH) FROM LopHocPhan
     WHERE MaHK IN (SELECT MaHK FROM HocKy
                    WHERE TrangThai = 'DangDienRa'))               AS MonHKNay,
    (SELECT COUNT(*) FROM vw_GPATichLuy
     WHERE GPATichLuy < 2.0)                                       AS SVCanChuY;
GO

-- ============================================================
--  STORED PROCEDURES
--  Tất cả SP đều tham chiếu ĐÚNG tên cột trong bảng/view
-- ============================================================
--  SP1: sp_DangNhap
--  Xác thực đăng nhập, cập nhật LanDangNhapCuoi
--  Trả về: MaND, VaiTro, ThanhCong (1/0), ThongBao
-- ----------------------------------------------------------
CREATE PROCEDURE sp_DangNhap
    @TenDangNhap NVARCHAR(50),
    @MatKhau     NVARCHAR(255)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @MaND   INT;
    DECLARE @VaiTro VARCHAR(15);

    SELECT @MaND = MaND, @VaiTro = VaiTro
    FROM   NguoiDung
    WHERE  TenDangNhap = @TenDangNhap
      AND  MatKhau     = @MatKhau
      AND  TrangThai   = 1;

    IF @MaND IS NOT NULL
    BEGIN
        UPDATE NguoiDung
        SET    LanDangNhapCuoi = GETDATE()
        WHERE  MaND = @MaND;

        SELECT @MaND   AS MaND,
               @VaiTro AS VaiTro,
               1       AS ThanhCong,
               N'Đăng nhập thành công' AS ThongBao;
    END
    ELSE
    BEGIN
        SELECT 0    AS MaND,
               NULL AS VaiTro,
               0    AS ThanhCong,
               N'Sai tên đăng nhập hoặc mật khẩu' AS ThongBao;

    @TenDangNhap NVARCHAR(50),
    @MatKhau     NVARCHAR(255),
    @Email       NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM NguoiDung WHERE TenDangNhap = @TenDangNhap)
    BEGIN
        SELECT -1 AS KetQua, N'Tên đăng nhập đã tồn tại' AS ThongBao;

        RETURN;
    END

    INSERT INTO NguoiDung (TenDangNhap, MatKhau, VaiTro, Email)
    VALUES (@TenDangNhap, @MatKhau, 'SinhVien', @Email);

    SELECT SCOPE_IDENTITY() AS MaND, N'Đăng ký thành công' AS ThongBao;
END;
GO

-- ----------------------------------------------------------
--  SP6: sp_NhapDiem
--  Dùng cho: NhapDiem button "Lưu điểm" và "Chốt điểm"
--  - Nếu chưa có bản ghi → INSERT
--  - Nếu đã có và chưa chốt → ghi lịch sử → UPDATE
--  - Nếu đã chốt → báo lỗi, không cho sửa
-- ----------------------------------------------------------
CREATE PROCEDURE sp_NhapDiem
    @MaDK       INT,
    @DiemCC     DECIMAL(4,2),
    @DiemGK     DECIMAL(4,2),
    @DiemCK     DECIMAL(4,2),
    @MaGV       VARCHAR(10),
    @ChotDiem   BIT           = 0,
    @LyDo       NVARCHAR(255) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    BEGIN TRY
        -- Lấy MaND của giảng viên để ghi vào LichSuDiem
        DECLARE @MaND INT;
        SELECT @MaND = MaND FROM GiangVien WHERE MaGV = @MaGV;

        IF @MaND IS NULL
        BEGIN
            SELECT 0 AS KetQua, N'Mã giảng viên không tồn tại' AS ThongBao;
            ROLLBACK TRANSACTION;
            RETURN;
        END

        IF EXISTS (SELECT 1 FROM Diem WHERE MaDK = @MaDK)
        BEGIN
            -- Kiểm tra đã chốt chưa
            IF EXISTS (SELECT 1 FROM Diem WHERE MaDK = @MaDK AND DaChotDiem = 1)
            BEGIN
                SELECT 0 AS KetQua, N'Điểm đã được chốt, không thể sửa' AS ThongBao;
                ROLLBACK TRANSACTION;
                RETURN;
            END

            -- Ghi lịch sử trước khi sửa
            INSERT INTO LichSuDiem (MaDiem, DiemCC_Cu, DiemGK_Cu, DiemCK_Cu, MaND_ThayDoi, LyDo)
            SELECT MaDiem, DiemCC, DiemGK, DiemCK, @MaND, @LyDo
            FROM   Diem
            WHERE  MaDK = @MaDK;

            -- Cập nhật điểm
            UPDATE Diem
            SET    DiemCC         = @DiemCC,
                   DiemGK         = @DiemGK,
                   DiemCK         = @DiemCK,
                   DaChotDiem     = @ChotDiem,
                   NgayNhap       = GETDATE(),
                   MaGV_NhapDiem  = @MaGV
            WHERE  MaDK = @MaDK;
        END
        ELSE
        BEGIN
            -- Chưa có bản ghi → INSERT mới
            INSERT INTO Diem (MaDK, DiemCC, DiemGK, DiemCK, DaChotDiem, MaGV_NhapDiem)
            VALUES (@MaDK, @DiemCC, @DiemGK, @DiemCK, @ChotDiem, @MaGV);
        END

        -- Nếu chốt điểm → cập nhật trạng thái đăng ký
        IF @ChotDiem = 1
        BEGIN
            UPDATE DangKyHocPhan
            SET    TrangThai = 'HoanThanh'
            WHERE  MaDK = @MaDK;
        END

        COMMIT TRANSACTION;
        SELECT 1 AS KetQua,
               CASE WHEN @ChotDiem = 1
                    THEN N'Chốt điểm thành công'
                    ELSE N'Lưu điểm thành công'
               END AS ThongBao;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        SELECT 0 AS KetQua, ERROR_MESSAGE() AS ThongBao;
    END CATCH
END;
GO

-- ----------------------------------------------------------
--  SP8: sp_ThongKe
--  Dùng cho: ThongKe form (lọc theo HK và Khoa)
--  Trả về 3 result set:
--    1. Tổng quan (tổng SV, môn, ĐTB, đạt, rớt)
--    2. Phân bố xếp loại (A/B+/B/C+/C/D+/D/F)
--    3. Tỉ lệ đạt/rớt theo từng môn
-- ----------------------------------------------------------
CREATE PROCEDURE sp_ThongKe
    @MaHK   VARCHAR(15) = NULL,
    @MaKhoa VARCHAR(10) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    -- Result Set 1: Tổng quan
    SELECT
        COUNT(DISTINCT dk.MaSV)                                        AS TongSV,
        COUNT(DISTINCT lhp.MaMH)                                       AS TongMon,
        ROUND(AVG(CAST(d.DiemTB AS FLOAT)), 2)                        AS DiemTBChung,
        SUM(CASE WHEN d.KetQua = 'Dat'      THEN 1 ELSE 0 END)       AS TongDat,
        SUM(CASE WHEN d.KetQua = 'KhongDat' THEN 1 ELSE 0 END)       AS TongRot
    FROM Diem d
    JOIN DangKyHocPhan dk ON d.MaDK    = dk.MaDK
    JOIN LopHocPhan lhp   ON dk.MaLHP  = lhp.MaLHP
    JOIN MonHoc mh        ON lhp.MaMH  = mh.MaMH
    WHERE dk.TrangThai = 'HoanThanh'
      AND (@MaHK   IS NULL OR lhp.MaHK  = @MaHK)
      AND (@MaKhoa IS NULL OR mh.MaKhoa = @MaKhoa);

    -- Result Set 2: Phân bố xếp loại
    SELECT
        d.XepLoai,
        COUNT(*) AS SoLuong
    FROM Diem d
    JOIN DangKyHocPhan dk ON d.MaDK    = dk.MaDK
    JOIN LopHocPhan lhp   ON dk.MaLHP  = lhp.MaLHP
    JOIN MonHoc mh        ON lhp.MaMH  = mh.MaMH
    WHERE dk.TrangThai = 'HoanThanh'
      AND (@MaHK   IS NULL OR lhp.MaHK  = @MaHK)
      AND (@MaKhoa IS NULL OR mh.MaKhoa = @MaKhoa)
    GROUP BY d.XepLoai
    ORDER BY d.XepLoai;

    -- Result Set 3: Tỉ lệ Đạt/Rớt theo môn
    SELECT
        MaMH, TenMH, SoDat, SoRot, DiemTBMon
    FROM vw_ThongKe_DatRot
    WHERE (@MaHK IS NULL OR MaHK = @MaHK);
END;
GO

    @MaMH       VARCHAR(10),

    @SoTinChi   TINYINT,
    @MaKhoa     VARCHAR(10),
    @LoaiMon    VARCHAR(15),

AS

    BEGIN
        IF EXISTS (SELECT 1 FROM MonHoc WHERE MaMH = @MaMH)
        BEGIN
            SELECT 0 AS KetQua, N'Mã môn học đã tồn tại' AS ThongBao;

-- ----------------------------------------------------------
--  SP11: sp_ThemSuaSinhVien
--  Dùng cho: ThemSuaSV form (button Lưu)
-- ----------------------------------------------------------
CREATE PROCEDURE sp_ThemSuaSinhVien
    @MaSV       VARCHAR(12),
    @HoTen      NVARCHAR(100),
    @NgaySinh   DATE          = NULL,
    @GioiTinh   VARCHAR(5)    = NULL,
    @DiaChi     NVARCHAR(200) = NULL,
    @SoDienThoai VARCHAR(15)  = NULL,
    @Email      NVARCHAR(100) = NULL,
    @MaLop      VARCHAR(15),
    @NamNhapHoc SMALLINT,
    @TinhTrang  VARCHAR(15)   = 'DangHoc',
    @TenDangNhap NVARCHAR(50) = NULL,
    @MatKhau    NVARCHAR(255) = NULL,
    @IsUpdate   BIT           = 0
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;
    BEGIN TRY
        IF @IsUpdate = 0
        BEGIN
            -- Tạo tài khoản NguoiDung trước
            DECLARE @MaND INT;
            INSERT INTO NguoiDung (TenDangNhap, MatKhau, VaiTro, Email)
            VALUES (
                ISNULL(@TenDangNhap, @MaSV),
                ISNULL(@MatKhau, @MaSV + '@123'),
                'SinhVien',
                @Email
            );
            SET @MaND = SCOPE_IDENTITY();

            INSERT INTO SinhVien
                (MaSV,MaND,HoTen,NgaySinh,GioiTinh,DiaChi,SoDienThoai,Email,MaLop,NamNhapHoc,TinhTrang)
            VALUES
                (@MaSV,@MaND,@HoTen,@NgaySinh,@GioiTinh,@DiaChi,@SoDienThoai,@Email,@MaLop,@NamNhapHoc,@TinhTrang);

            COMMIT TRANSACTION;
            SELECT 1 AS KetQua, N'Thêm sinh viên thành công' AS ThongBao;
        END
        ELSE
        BEGIN
            UPDATE SinhVien
            SET HoTen=@HoTen, NgaySinh=@NgaySinh, GioiTinh=@GioiTinh,
                DiaChi=@DiaChi, SoDienThoai=@SoDienThoai, Email=@Email,
                MaLop=@MaLop, NamNhapHoc=@NamNhapHoc, TinhTrang=@TinhTrang
            WHERE MaSV = @MaSV;

            COMMIT TRANSACTION;
            SELECT 1 AS KetQua, N'Cập nhật sinh viên thành công' AS ThongBao;
        END
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        SELECT 0 AS KetQua, ERROR_MESSAGE() AS ThongBao;
    END CATCH
END;
GO
